-- =============================================================================
-- PROJETO DE GESTÃO ESCOLAR - DDL OFICIAL (POSTGRESQL)
-- =============================================================================

-- Limpeza preventiva de tabelas dependentes caso precise recriar do zero
DROP TABLE IF EXISTS controle_academico CASCADE;
DROP TABLE IF EXISTS aulas CASCADE;
DROP TABLE IF EXISTS temas CASCADE;
DROP TABLE IF EXISTS alunos CASCADE;
DROP TABLE IF EXISTS professores CASCADE;
DROP TABLE IF EXISTS administradores CASCADE;
DROP TABLE IF EXISTS eventos_agenda CASCADE;
DROP TABLE IF EXISTS niveis CASCADE;
DROP TABLE IF EXISTS usuarios CASCADE;

-- =============================================================================
-- 1. TABELAS DE AUTENTICAÇÃO E DOMÍNIO BASE
-- =============================================================================

-- Autenticação unificada (Entidade Pai)
CREATE TABLE usuarios (
    id SERIAL PRIMARY KEY,
    email VARCHAR(100) UNIQUE NOT NULL,
    senha VARCHAR(255) NOT NULL,
    perfil VARCHAR(20) NOT NULL -- 'ADMIN', 'PROFESSOR', 'ALUNO'
);

-- Tabela de domínio/lookup para Níveis Pedagógicos (3ª Forma Normal)
CREATE TABLE niveis (
    id_nivel SERIAL PRIMARY KEY,
    sigla VARCHAR(20) UNIQUE NOT NULL, -- 'BASICO', 'INTERMEDIARIO', 'AVANCADO'
    nome VARCHAR(50) NOT NULL,
    descricao TEXT
);

-- =============================================================================
-- 2. PERFIS ESPECÍFICOS DE USUÁRIO
-- =============================================================================

-- Secretaria / Direção (Acesso global administrativo)
CREATE TABLE administradores (
    id_admin SERIAL PRIMARY KEY,
    id_usuario INT UNIQUE NOT NULL,
    nome VARCHAR(100) NOT NULL,
    cargo VARCHAR(50) DEFAULT 'SECRETARIA',
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id) ON DELETE CASCADE
);

-- Corpo docente (Gestão de aulas, temas e lançamento de notas/presença)
CREATE TABLE professores (
    id_professor SERIAL PRIMARY KEY,
    id_usuario INT UNIQUE NOT NULL,
    nome VARCHAR(100) NOT NULL,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id) ON DELETE CASCADE
);

-- Alunos (Campos de endereço estruturados para integração com ViaCEP)
CREATE TABLE alunos (
    id_aluno SERIAL PRIMARY KEY,
    id_usuario INT UNIQUE NOT NULL,
    id_nivel INT NOT NULL DEFAULT 1,
    nome VARCHAR(100) NOT NULL,
    cpf VARCHAR(14) UNIQUE,
    cep VARCHAR(9) NOT NULL,
    logradouro VARCHAR(150) NOT NULL,
    bairro VARCHAR(100) NOT NULL,
    cidade VARCHAR(100) NOT NULL,
    uf CHAR(2) NOT NULL,
    numero VARCHAR(20) NOT NULL,
    complemento VARCHAR(100),
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (id_nivel) REFERENCES niveis(id_nivel) ON DELETE RESTRICT
);

-- =============================================================================
-- 3. PLANEJAMENTO PEDAGÓGICO E AULAS INDIVIDUAIS
-- =============================================================================

-- Conteúdos/Temas didáticos vinculados ao nível acadêmico
CREATE TABLE temas (
    id_tema SERIAL PRIMARY KEY,
    id_nivel INT NOT NULL,
    titulo_tema VARCHAR(150) NOT NULL,
    descricao TEXT,
    FOREIGN KEY (id_nivel) REFERENCES niveis(id_nivel) ON DELETE RESTRICT
);

-- Aulas individuais com slot padronizado de 1 hora e anexos em /static/uploads/
CREATE TABLE aulas (
    id_aula SERIAL PRIMARY KEY,
    id_professor INT NOT NULL,
    id_aluno INT NOT NULL,
    id_tema INT, -- Opcional/flexível
    titulo_aula VARCHAR(150) NOT NULL,
    data_aula DATE NOT NULL,
    horario_inicio TIME NOT NULL, -- Duração presumida de 1 hora
    link_aula VARCHAR(255),
    status VARCHAR(20) DEFAULT 'AGENDADA', -- 'AGENDADA', 'CONCLUIDA', 'CANCELADA', 'FERIAS'
    url_pdf VARCHAR(255),                  -- Caminho do material salvo em /static/uploads/
    url_slides VARCHAR(255),               -- Caminho dos slides salvos em /static/uploads/
    observacoes TEXT,
    CONSTRAINT uk_professor_agenda UNIQUE (id_professor, data_aula, horario_inicio),
    FOREIGN KEY (id_professor) REFERENCES professores(id_professor) ON DELETE RESTRICT,
    FOREIGN KEY (id_aluno) REFERENCES alunos(id_aluno) ON DELETE CASCADE,
    FOREIGN KEY (id_tema) REFERENCES temas(id_tema) ON DELETE SET NULL
);

-- =============================================================================
-- 4. CONTROLE ACADÊMICO E AUTOAVALIAÇÃO
-- =============================================================================

-- Registro de chamada, autoavaliação e conclusão da aula pelo aluno
CREATE TABLE controle_academico (
    id_controle SERIAL PRIMARY KEY,
    id_aluno INT NOT NULL,
    id_aula INT NOT NULL,
    presenca BOOLEAN DEFAULT FALSE,       -- Controle de presença do Professor
    aula_assistida BOOLEAN DEFAULT FALSE, -- Confirmação marcada pelo Aluno
    data_conclusao TIMESTAMP,             -- Momento da conclusão pelo aluno
    nota_professor DECIMAL(5,2),          -- Nota atribuída pelo Professor
    nota_autoav DECIMAL(5,2),             -- Nota de autoavaliação do Aluno
    anotacoes_aluno TEXT,
    CONSTRAINT uk_aluno_aula UNIQUE (id_aluno, id_aula),
    FOREIGN KEY (id_aluno) REFERENCES alunos(id_aluno) ON DELETE CASCADE,
    FOREIGN KEY (id_aula) REFERENCES aulas(id_aula) ON DELETE CASCADE
);

-- =============================================================================
-- 5. MURAL INSTITUCIONAL E COMUNICADOS
-- =============================================================================

-- Avisos, comunicados e eventos da escola (integrados aos feriados no calendário)
CREATE TABLE eventos_agenda (
    id_evento SERIAL PRIMARY KEY,
    titulo VARCHAR(150) NOT NULL,
    descricao TEXT,
    data_evento DATE NOT NULL,
    tipo_evento VARCHAR(50) DEFAULT 'AVISO' -- 'AVISO', 'PROVA', 'REUNIAO'
);
